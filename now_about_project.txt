```python
import hashlib
import json

def generate_hash(row_dict: dict) -> str:
    row_str = json.dumps(row_dict, sort_keys=True)
```

### Scheduled Job

```python
import pandas as pd
from sqlalchemy.orm import Session
from database import SessionLocal
from models import TenderData

def flatten_json(data: dict) -> str:
    return " ".join(str(v) for v in data.values() if v)

def sync_excel(tender_id: int, excel_path: str):
    db: Session = SessionLocal()

    df = pd.read_excel(excel_path)

    for idx, row in df.iterrows():
        row_dict = row.to_dict()
        row_hash = generate_hash(row_dict)

        record = TenderData(
            tender_id=tender_id,
            row_no=idx + 1,
            row_hash=row_hash,
            dynamic_data=row_dict,
            search_text=flatten_json(row_dict)
        )

        db.merge(record)  # UPSERT
    db.commit()
    db.close()
```

```python
import httpx
import logging
from io import BytesIO

logger = logging.getLogger("onedrive")

class OneDriveService:

    def __init__(self, client_id, client_secret, tenant_id):
        self.client_id = client_id
        self.client_secret = client_secret
        self.tenant_id = tenant_id
        self.token_url = f"https://login.microsoftonline.com/{tenant_id}/oauth2/v2.0/token"

    async def get_access_token(self):
        data = {
            "client_id": self.client_id,
            "scope": "https://graph.microsoft.com/.default",
            "client_secret": self.client_secret,
            "grant_type": "client_credentials",
        }
        async with httpx.AsyncClient(timeout=30) as client:
            response = await client.post(self.token_url, data=data)
        response.raise_for_status()
        return response.json()["access_token"]

    async def download_excel(self, item_id: str):
        token = await self.get_access_token()
        url = f"https://graph.microsoft.com/v1.0/me/drive/items/{item_id}/content"

        headers = {"Authorization": f"Bearer {token}"}
        async with httpx.AsyncClient(timeout=60) as client:
            response = await client.get(url, headers=headers)
        response.raise_for_status()

        return BytesIO(response.content)
```

## **2️⃣ Generic scraper logic**

Here’s the approach:

1. Open site.
2. Fill fields based on `type` and `label`.
3. Select dropdown options by label/value.
4. Click radio buttons by label.
5. Click search button.
6. Scrape table(s).
7. Save results to JSON.

Key point: **the code never relies on fragile IDs or classes unless you provide them as fallback**.

Here’s a fully generic Playwright example:

```javascript
const { chromium } = require('playwright');
const fs = require('fs');

(async () => {
  let sites = [];
  try {
    sites = JSON.parse(fs.readFileSync('sites.json', 'utf-8'));
  } catch (err) {
    console.error("Error: sites.json not found or invalid!");
    process.exit(1);
  }

  const browser = await chromium.launch({ headless: false });
  const page = await browser.newPage();

  for (const site of sites) {
    console.log(`Opening ${site.siteName}...`);
    await page.goto(site.link, { waitUntil: 'domcontentloaded' });

    // Fill fields
    for (const field of site.fields) {
      if (!field.label || !field.type) continue;

      try {
        switch (field.type.toLowerCase()) {
          case 'text':
            await page.fill(`input[placeholder="${field.label}"], input[aria-label="${field.label}"]`, field.value || '');
            console.log(`Filled "${field.label}"`);
            break;

          case 'radio':
            await page.check(`//label[contains(text(),"${field.label}")]/preceding-sibling::input[@type='radio']`);
            console.log(`Selected radio "${field.label}"`);
            break;

          case 'select':
          case 'option':
            await page.selectOption(`select[aria-label="${field.label}"], select[name="${field.label}"]`, { label: field.value });
            console.log(`Selected "${field.value}" for "${field.label}"`);
            break;

          default:
            console.log(`Unknown type "${field.type}"`);
        }
      } catch (err) {
        console.log(`Could not fill/select "${field.label}": ${err.message}`);
      }
    }

    // Click search button
    if (site.searchButton && site.searchButton.label) {
      try {
        await page.click(`//button[contains(text(),"${site.searchButton.label}")] | //input[@type='submit' and @value="${site.searchButton.label}"]`);
        console.log(`Clicked search button: ${site.searchButton.label}`);
      } catch (err) {
        console.log(`Could not click search button "${site.searchButton.label}": ${err.message}`);
      }
    }

    await page.waitForTimeout(3000); // wait for table to render

    // Scrape table
    let tableData = [];
    try {
      const selector = site.tableSelector || 'table';
      const tableExists = await page.$(selector);
      if (tableExists) {
        tableData = await page.$$eval(`${selector} tr`, rows =>
          rows.map(r => Array.from(r.querySelectorAll('th, td'), td => td.innerText.trim()))
        );
      }
      console.log(`Scraped ${tableData.length} rows from ${site.siteName}`);
      fs.writeFileSync(`${site.siteName.replace(/\s+/g,'_')}_bids.json`, JSON.stringify(tableData, null, 2));
    } catch (err) {
      console.log(`Error scraping table: ${err.message}`);
    }
  }

  await browser.close();
})();
```
