### Table Requirement (Unique row key)

Add a hash or unique key per row:


### Hash function (Detect changes)

```python
import hashlib
import json

def generate_hash(row_dict: dict) -> str:
    row_str = json.dumps(row_dict, sort_keys=True)
    return hashlib.sha256(row_str.encode()).hexdigest()
```

---

### Scheduled Job

```python
import pandas as pd
from sqlalchemy.orm import Session
from database import SessionLocal
from models import TenderData

def flatten_json(data: dict) -> str:
    return " ".join(str(v) for v in data.values() if v)

def sync_excel(tender_id: int, excel_path: str):
    db: Session = SessionLocal()

    df = pd.read_excel(excel_path)

    for idx, row in df.iterrows():
        row_dict = row.to_dict()
        row_hash = generate_hash(row_dict)

        record = TenderData(
            tender_id=tender_id,
            row_no=idx + 1,
            row_hash=row_hash,
            dynamic_data=row_dict,
            search_text=flatten_json(row_dict)
        )

        db.merge(record)  # UPSERT
    db.commit()
    db.close()
```
